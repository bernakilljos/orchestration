1일 비용 분석 쿼리

SELECT ai, COUNT(*) as calls,
        ROUND(SUM(cost_usd), 2)
        as cost,
        ROUND(AVG(latency_ms))
        as avg_ms
FROM metrics
WHERE recorded_at >
    unixepoch('now', '-1 day')
GROUP BY ai
ORDER BY cost DESC;
